# Phase 1 push 안내 — Windows 셸에서 실행

cd "D:\훙냥냥\마렌상\roll20-sheet-builder 시트 고치기\web"

# 1. node_modules 설치 (Windows 에선 pnpm 정상 동작)
pnpm install

# 2. 로컬 동작 확인
pnpm dev
# → http://localhost:3000 에서 Blockly workspace 확인 후 Ctrl+C

# 3. 정적 빌드 확인 (선택)
pnpm build
# → out/ 디렉토리 생성

# 4. git 초기화 + push
git init -b main
git add .
git commit -m "Phase 1: Next.js + Blockly bootstrap"
git remote add origin https://github.com/Song991123/roll20-block-editor.git
git push -u origin main

# 5. GitHub Pages 활성화
